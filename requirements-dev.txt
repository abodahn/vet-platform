# Test tooling. Install alongside requirements.txt:
#   pip install -r requirements.txt -r requirements-dev.txt
#
# Everything else tests/ imports (psycopg2, requests) is already a runtime
# dependency, and the rest is stdlib. Playwright is deliberately absent — the
# browser suite lives in ../tests and is not part of CI.

pytest>=8.0.0

# An embedded PostgreSQL server, so tests/test_postgres_full.py can run against
# the real production engine on a machine with no PostgreSQL installed. Ships
# its own binaries inside the virtualenv: no admin rights, no Windows service,
# nothing outside .venv, and `pip uninstall pgembed` removes it entirely.
#
# Worth the dependency because the suite it enables cannot be replaced by the
# SQLite run. It found four bugs that exist ONLY on PostgreSQL, including one
# where a wrong password raised UndefinedColumn instead of being reported as a
# wrong password, leaving account lockout permanently disengaged.
#
# See scripts/run_postgres_tests.py for the one-command version.
pgembed>=0.2.0

# Reads back the PDFs the app writes, so a test can assert that a medical
# history report actually contains the diagnosis and that no Arabic letter was
# dropped on the way to paper.
#
# This replaced ~90 lines of hand-rolled PDF in tests/test_crm_routes.py:
# inflate the page streams, scrape /ToUnicode CMaps, undo string escaping, map
# glyph ids back to characters. It decoded every glyph run with every font's
# map and accepted any result that matched, so the wrong subset's numbering
# produced plausible garbage -- a test passed for a year on text the PDF never
# contained, then failed claiming letters were being dropped from medical
# records when nothing was wrong. A test helper that can raise a false alarm
# about a patient record has negative value. pypdf is pure Python, has no
# transitive dependencies, and is right.
pypdf>=5.0.0
